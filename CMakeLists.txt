cmake_minimum_required(VERSION 3.16)
project(VitalWatch VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(vitalwatch_core
    src/SensorSimulator.cpp src/ReadingValidator.cpp src/AlertEngine.cpp
    src/EventLogger.cpp src/Replay.cpp src/Pipeline.cpp)
target_include_directories(vitalwatch_core PUBLIC include)
target_link_libraries(vitalwatch_core PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(vitalwatch_core PUBLIC /W4 /permissive- /utf-8)
else()
    target_compile_options(vitalwatch_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
add_executable(vitalwatch src/main.cpp)
target_link_libraries(vitalwatch PRIVATE vitalwatch_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(vitalwatch_tests tests/tests.cpp)
    target_link_libraries(vitalwatch_tests PRIVATE vitalwatch_core)
    foreach(case IN ITEMS valid invalid_hr invalid_oxygen invalid_temperature threshold
        missing disconnected csv shutdown malformed deterministic boundaries config
        producer_error consumer_error queue_close missing_file replay config_invalid nonfinite)
        add_test(NAME ${case} COMMAND vitalwatch_tests ${case})
        set_tests_properties(${case} PROPERTIES TIMEOUT 10)
    endforeach()
    add_test(NAME cli_demo COMMAND vitalwatch --deterministic --count 8 --interval-ms 0 --log-dir cli-logs)
    set_tests_properties(cli_demo PROPERTIES PASS_REGULAR_EXPRESSION "Processed 8 readings" TIMEOUT 10)
    add_test(NAME cli_invalid COMMAND vitalwatch --count abc)
    set_tests_properties(cli_invalid PROPERTIES WILL_FAIL TRUE TIMEOUT 10)
    add_test(NAME cli_missing_argument COMMAND vitalwatch --replay)
    set_tests_properties(cli_missing_argument PROPERTIES WILL_FAIL TRUE TIMEOUT 10)
endif()
