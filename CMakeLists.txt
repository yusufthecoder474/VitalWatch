cmake_minimum_required(VERSION 3.16)
project(VitalWatch VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
option(VITALWATCH_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
option(VITALWATCH_ENABLE_ASAN "Enable AddressSanitizer (GCC/Clang or MSVC)" OFF)
option(VITALWATCH_ENABLE_UBSAN "Enable UndefinedBehaviorSanitizer (GCC/Clang only)" OFF)
set(VITALWATCH_SANITIZER "none" CACHE STRING "Legacy profile: none, asan, asan-ubsan, ubsan, tsan")
set_property(CACHE VITALWATCH_SANITIZER PROPERTY STRINGS none asan asan-ubsan ubsan tsan)
add_library(vitalwatch_core
    src/SensorSimulator.cpp src/ReadingValidator.cpp src/AlertEngine.cpp
    src/EventLogger.cpp src/Replay.cpp src/Pipeline.cpp)
target_include_directories(vitalwatch_core PUBLIC include)
target_link_libraries(vitalwatch_core PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(vitalwatch_core PUBLIC /W4 /permissive- /utf-8)
else()
    target_compile_options(vitalwatch_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
if(VITALWATCH_WARNINGS_AS_ERRORS)
    if(MSVC)
        target_compile_options(vitalwatch_core PUBLIC /WX)
    else()
        target_compile_options(vitalwatch_core PUBLIC -Werror)
    endif()
endif()
if(NOT VITALWATCH_SANITIZER MATCHES "^(none|asan|asan-ubsan|ubsan|tsan)$")
    message(FATAL_ERROR "Unknown VITALWATCH_SANITIZER: ${VITALWATCH_SANITIZER}")
endif()
set(use_asan ${VITALWATCH_ENABLE_ASAN})
set(use_ubsan ${VITALWATCH_ENABLE_UBSAN})
if(VITALWATCH_SANITIZER MATCHES "^asan")
    set(use_asan ON)
endif()
if(VITALWATCH_SANITIZER STREQUAL "ubsan" OR VITALWATCH_SANITIZER STREQUAL "asan-ubsan")
    set(use_ubsan ON)
endif()
if(VITALWATCH_SANITIZER STREQUAL "tsan" AND (use_asan OR use_ubsan))
    message(FATAL_ERROR "Use a separate TSan build; ASan/UBSan options cannot be combined with this profile")
endif()
if(MSVC)
    if(use_ubsan OR VITALWATCH_SANITIZER STREQUAL "tsan")
        message(FATAL_ERROR "MSVC supports this project's ASan option only; UBSan/TSan require GCC/Clang")
    endif()
    if(use_asan)
        if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC" AND MSVC_VERSION LESS 1930)
            message(FATAL_ERROR "This project's MSVC ASan recipe targets VS 2022 or newer")
        endif()
        # ASan is incompatible with /RTC, Edit-and-Continue and incremental linking.
        foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
            string(REGEX REPLACE "/RTC[1su]+" "" CMAKE_CXX_FLAGS_${config} "${CMAKE_CXX_FLAGS_${config}}")
            string(REPLACE "/ZI" "/Zi" CMAKE_CXX_FLAGS_${config} "${CMAKE_CXX_FLAGS_${config}}")
        endforeach()
        message(STATUS "MSVC ASan: config flags omit /RTC and use /Zi instead of /ZI")
        target_compile_options(vitalwatch_core PUBLIC /fsanitize=address)
        target_link_options(vitalwatch_core PUBLIC /INCREMENTAL:NO /DEBUG:FULL)
    endif()
elseif(use_asan OR use_ubsan OR VITALWATCH_SANITIZER STREQUAL "tsan")
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Sanitizers require a compatible GCC, Clang or MSVC runtime")
    endif()
    if(VITALWATCH_SANITIZER STREQUAL "tsan")
        set(sanitizer_flag -fsanitize=thread)
    elseif(use_asan AND use_ubsan)
        set(sanitizer_flag -fsanitize=address,undefined)
    elseif(use_asan)
        set(sanitizer_flag -fsanitize=address)
    else()
        set(sanitizer_flag -fsanitize=undefined)
    endif()
    target_compile_options(vitalwatch_core PUBLIC ${sanitizer_flag} -fno-omit-frame-pointer -fno-sanitize-recover=all)
    target_link_options(vitalwatch_core PUBLIC ${sanitizer_flag})
endif()
add_executable(vitalwatch src/main.cpp)
target_link_libraries(vitalwatch PRIVATE vitalwatch_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(vitalwatch_tests tests/tests.cpp)
    target_link_libraries(vitalwatch_tests PRIVATE vitalwatch_core)
    foreach(case IN ITEMS valid invalid_hr invalid_oxygen invalid_temperature threshold
        missing disconnected csv shutdown malformed deterministic boundaries config
        producer_error consumer_error queue_close missing_file replay config_invalid nonfinite)
        add_test(NAME ${case} COMMAND vitalwatch_tests ${case})
        set_tests_properties(${case} PROPERTIES TIMEOUT 10)
    endforeach()
    foreach(mode IN ITEMS demo invalid missing_argument)
        add_test(NAME cli_${mode} COMMAND ${CMAKE_COMMAND}
            -DPROGRAM=$<TARGET_FILE:vitalwatch> -DMODE=${mode}
            -DWORK_DIR=${CMAKE_CURRENT_BINARY_DIR}/cli-${mode}
            -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/verify_cli.cmake)
        set_tests_properties(cli_${mode} PROPERTIES TIMEOUT 10)
    endforeach()
    add_executable(vitalwatch_concurrency_tests tests/concurrency_tests.cpp)
    target_link_libraries(vitalwatch_concurrency_tests PRIVATE vitalwatch_core)
    target_compile_definitions(vitalwatch_concurrency_tests PRIVATE VITALWATCH_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")
    add_executable(vitalwatch_firmware_tests tests/firmware_tests.cpp)
    target_link_libraries(vitalwatch_firmware_tests PRIVATE vitalwatch_core)
    foreach(case IN ITEMS fw_malformed fw_active_shutdown fw_capacity fw_close_saturated fw_bounded_exception fw_replay_mismatch)
        add_test(NAME ${case} COMMAND vitalwatch_firmware_tests ${case})
        set_tests_properties(${case} PROPERTIES TIMEOUT 10)
    endforeach()
    foreach(case IN ITEMS empty_pipeline one_sample large_fifo sequence_audit queue_wait_close
            queued_drain exact_worker_errors deterministic_replay)
        add_test(NAME ${case} COMMAND vitalwatch_concurrency_tests ${case})
        set_tests_properties(${case} PROPERTIES TIMEOUT 10)
    endforeach()
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        add_test(NAME stdout_error COMMAND ${CMAKE_COMMAND}
            -DPROGRAM=$<TARGET_FILE:vitalwatch> -DWORK_DIR=${CMAKE_CURRENT_BINARY_DIR}/stdout-error
            -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/verify_stdout.cmake)
        set_tests_properties(stdout_error PROPERTIES TIMEOUT 10)
        find_program(VITALWATCH_BASH bash)
        if(VITALWATCH_BASH)
            add_test(NAME signal_shutdown COMMAND ${VITALWATCH_BASH}
                ${CMAKE_CURRENT_SOURCE_DIR}/tools/check-signal.sh $<TARGET_FILE:vitalwatch>
                ${CMAKE_CURRENT_BINARY_DIR}/signal-test)
            set_tests_properties(signal_shutdown PROPERTIES TIMEOUT 10)
        endif()
    endif()
endif()
